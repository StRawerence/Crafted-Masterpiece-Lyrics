[Intro]
ذهبٌ ضائعٌ يعودُ إلى التُراب
وبابٌ أخيرٌ للمدينةِ صارَ خَراب
في كفّي فراغٌ لا سُلطانَ ولا أسباب
وكلامي إذا ضاقَ الزمانُ خفَّ حِساب

[Verse]
مجدٌ منسيٌّ خلفي بلا دار
وحقٌّ عتيقٌ دُفِنَ تحتَ أستار
أشربُ قَسَمَ الوحدةِ صبرًا وإصرار
وأسألُ السَّماءَ أيُّ دربٍ إلى مَسار

[Pre-Chorus]
سِترُ النُّجومِ يُحصيني بلا كلام
والرملُ يُطاردُ خُطواتي بلا نِظام
وأنا أثبتُ ولا يسقطُ فيَّ العَزم

[Chorus]
يا ميثاقَ المَقامِ الأسمى لا تذبل
لا أبتغي مُعجزةً ولا وعدًا ينهمِل
لتحفظِ الأرضُ أثري فوقَ المدى
وليَبقَ للعالمِ آخرُ نِظامٍ وهُدى
ما دُمتُ أمشي ولا يخونُ القَسَم

[Verse]
شمسٌ قاسيةٌ تُلهبُ حلقي بوجع
وسَرابٌ ينسجُ للقلبِ وعدًا خَدَع
همسُ الإغراءِ يأمُرني بأن أرجع
ودمي في صدري يسألُ أينَ المُنْعَطَف

[Pre-Chorus]
الجسدُ ينتهي ويرجعُ للغُبار
والروحُ تعبرُ وقتًا فوقَ الأعمار
وأنا أتقِنُ خُطوتي ضِدَّ الانكِسار

[Chorus]
أقسمُ للناسِ وأُثبّتُهُ بالعَناء
والصحراءُ تبلعُ صوتي ولا تدعُ نِداء
والنُّورُ البعيدُ شاهدٌ لا ينام
والسَّماءُ ختمٌ يعُدُّ وحدتي في المَقام
وإن غابَ اسمي يبقى الأثرُ في دَوام

[Bridge]
لا أحتاجُ صرحًا عاليًا ليفهمَ العهد
يكفي أن يسكنَ في العظمِ فيصيرَ أشد
ويرتجفَ الصوتُ بينَ الشُّقوقِ ثم يشتد
أُسلّمُهُ لفجرٍ قادمٍ يومًا يلِد

[Chorus]
أُسلّمُ القَسَمَ لغدٍ صافٍ وثَبات
ولتعبُر إرادةُ العُلوِّ الذي لا يُسمّى كلَّ حدٍّ وجِهات
لا تلتفت حتى يكمُلَ في القلبِ السُّكون
لا تتردّد حتى يصيرَ الطريقُ يقين
لا تشكّ حتى يستقيمَ العالمُ ويلين
أريدُ أن أبقى مجهولًا وبلا قبرٍ وبلا عُنوان
ولا أطلبُ إلّا السَّلام

[Outro]
سلامٌ للناسِ كلّهم في كلِّ مكان
سلامٌ للأرضِ حتى يهدأَ كلُّ لِسان
سلامٌ لكلِّ قلبٍ تعِبَ من الخُسران
سلامٌ لكلِّ دربٍ يحملُ فيهِ الإنسان
وشكرٌ للسَّلامِ إذا صارَ ميزان
